Cloud sync (Supabase)

Saves every practice to your team's Supabase project: the practice itself (name, date, video link), one row per possession (start / end time) and one row per stat (timestamp, player, possession). Every coach who connects with the same key sees the same list in the Practices tab.

Setup SQL — run once in Supabase → SQL Editor (safe to re-run) 
-- One row per practice (full data, used to reopen it in the app)
create table if not exists public.sessions (
  id text primary key,
  name text,
  source jsonb,                  -- video link: {"type":"yt","id":…} or {"type":"file","name":…}
  data jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.sessions add column if not exists practice_date date;

-- One row per possession: start and end time in the video
create table if not exists public.possessions (
  session_id text not null references public.sessions(id) on delete cascade,
  n int not null,                -- possession number in the practice
  team text not null,            -- GOLD / BLUE
  team_n int,                    -- that team's possession number
  start_s double precision not null,
  end_s double precision,
  points int not null default 0,
  video_url text,                -- YouTube link at the possession start (null for local files)
  primary key (session_id, n)
);

-- The roster: names live only here, so only people with the project key can see them
create table if not exists public.players (
  id text primary key,           -- what stats point at (Player 1…10 for the original placeholders)
  name text not null,
  number text,
  active boolean not null default true,
  sort int,
  updated_at timestamptz not null default now()
);

-- One row per coded stat / shot / free throw, with its timestamp
create table if not exists public.events (
  session_id text not null references public.sessions(id) on delete cascade,
  id int not null,
  possession_n int,              -- which possession it happened in
  t double precision not null,   -- seconds into the video
  kind text not null,            -- stat / shot / ft
  stat text,                     -- AST, TO, DREB, … ; shot label; FT
  player text,                   -- players.id
  team text,
  make boolean,
  zone text, grade text,
  x double precision, y double precision,
  video_url text,                -- YouTube link a few seconds before the moment
  primary key (session_id, id)
);
create index if not exists events_player_stat on public.events (player, stat);

alter table public.sessions enable row level security;
alter table public.possessions enable row level security;
alter table public.events enable row level security;
alter table public.players enable row level security;
drop policy if exists "team read/write" on public.sessions;
drop policy if exists "team read/write" on public.possessions;
drop policy if exists "team read/write" on public.events;
drop policy if exists "team read/write" on public.players;
create policy "team read/write" on public.sessions for all to anon, authenticated using (true) with check (true);
create policy "team read/write" on public.possessions for all to anon, authenticated using (true) with check (true);
create policy "team read/write" on public.events for all to anon, authenticated using (true) with check (true);
create policy "team read/write" on public.players for all to anon, authenticated using (true) with check (true);
grant select, insert, update, delete on public.sessions, public.possessions, public.events, public.players to anon, authenticated;

-- Make the API pick up the new tables right away
notify pgrst, 'reload schema';
These policies let anyone with your project key read and write — fine for a staff tool, but don't post the key publicly.
